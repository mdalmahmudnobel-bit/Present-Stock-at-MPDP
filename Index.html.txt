function call(name, ...args) {
  return new Promise(async (resolve, reject) => {
    try {
      if (!APPS_SCRIPT_URL || APPS_SCRIPT_URL === 'PASTE_YOUR_APPS_SCRIPT_WEB_APP_URL_HERE') {
        throw new Error('Set APPS_SCRIPT_URL to your deployed Google Apps Script Web App (/exec) URL first.');
      }
      const res = await fetch(APPS_SCRIPT_URL, {
        method: 'POST',
        headers: { 'Content-Type': 'text/plain;charset=utf-8' },
        body: JSON.stringify({ action: name, args: args })
      });
      const text = await res.text();
      let json;
      try {
        json = JSON.parse(text);
      } catch (_) {
        throw new Error('Invalid response from Apps Script Web App. Check deployment access ("Anyone").');
      }
      if (!json || !json.ok) {
        throw new Error((json && json.error) || 'Server request failed.');
      }
      resolve(json.data);
    } catch (err) {
      reject(new Error(err && err.message ? err.message : String(err || 'Network error')));
    }
  });
}

async function submitPosting(){
  const b=document.querySelector('button[onclick="submitPosting()"]');
  try{
    clearFieldErrors();
    const common={challanDate:$('challanDate').value,receivedDate:$('receivedDate').value,mrr:$('mrr').value,pr:$('pr').value,vendor:$('vendor').value.trim(),challanNo:$('challanNo').value.trim(),equip:$('equip').value.trim()};
    if(!common.challanDate){showFieldError($('challanDate'),'Challan Date is required. Please select the Challan Date.');return}
    if(!common.receivedDate){showFieldError($('receivedDate'),'Received Date is required. Please select the Received Date.');return}
    if(!common.vendor){showFieldError($('vendor'),'Vendor Name is required. Please enter the Vendor Name.');return}
    if(!common.challanNo){showFieldError($('challanNo'),'Challan No is required. Please enter the Challan No.');return}
    if(!common.equip){showFieldError($('equip'),'Equip is required. Please enter the Equipment name.');return}
    const enteredRows=[...$('items').children].map((el,index)=>{let x={};el.querySelectorAll('[data-k]').forEach(i=>x[i.dataset.k]=i.value.trim());return {row:el,index:index+1,data:x}}).filter(x=>Object.values(x.data).some(v=>String(v||'').trim()!==''));
    if(!enteredRows.length){msg('postMsg','Please enter at least one Item Name.',true);const first=$('items').querySelector('[data-k="item"]');if(first){first.classList.add('field-invalid');first.focus()}return}
    for(const entry of enteredRows){
      const x=entry.data;
      if(!x.item){showFieldError(entry.row.querySelector('[data-k="item"]'),'Item Name is required in row '+entry.index+'.');return}
      if(!x.unit){showFieldError(entry.row.querySelector('[data-k="unit"]'),'Unit is required for item row '+entry.index+'.');return}
      if(!x.qty||Number(x.qty)<=0){showFieldError(entry.row.querySelector('[data-k="qty"]'),'Quantity is required and must be greater than zero in row '+entry.index+'.');return}
    }
    const items=enteredRows.map(x=>x.data);
    if($('refLookup').value.trim()&&!editingPostingCode){msg('postMsg','Please load the entered REF first. Existing records will then be updated instead of creating a new posting.',true);return}
    b.disabled=true;b.textContent='Saving...';
    const r=await call('postPosting',token,{common,items,postingCode:editingPostingCode});
    $('success').style.display='block';
    $('successCode').textContent=r.postingCode;
    updateLastSubmitted(r.postingCode);
    $('successInfo').textContent='Items saved: '+r.itemCount;
    renderHistory(r.history);
    clearPostingForm();
    msg('postMsg',(r.mode==='updated'?'Updated existing ':'Saved new ')+r.postingCode+' successfully. Form cleared for the next posting.');
    await refreshSuggestions();
  }catch(e){msg('postMsg',e.message,true)}
  finally{if(b){b.disabled=false;b.textContent='Submit & Save to Google Sheet'}}
}
